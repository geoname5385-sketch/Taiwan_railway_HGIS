[...allFeatures] // 如果沒打字，就抓出全部 : allFeatures.filter(f => { if (!f.properties) return false; const nameMatch = normalize(f.properties.name).includes(cleanKeyword); const tailoMatch = normalize(f.properties.tailo || "").includes(cleanKeyword); return nameMatch || tailoMatch; }); // 2. 強制排序：依照 tailo (台羅拼音) 的 A-Z 字母順序排列 matched.sort((a, b) => { const tailoA = a.properties.tailo || ""; const tailoB = b.properties.tailo || ""; return tailoA.localeCompare(tailoB, 'en'); }); if (keyword === '') { if(geojsonLayer) geojsonLayer.setStyle({ color: "#ff7800", weight: 4, opacity: 0.8 }); } if (matched.length > 0) { dropdown.style.display = 'block'; matched.forEach(f => { const item = document.createElement('div'); item.className = 'search-item'; // 3. 視覺顯示：只呈現漢字 name，保持介面清爽 item.textContent = f.properties.name; item.onclick = () => { dropdown.style.display = 'none'; input.value = f.properties.name; clearBtn.style.display = 'block'; geojsonLayer.eachLayer(layer => { if (layer.feature === f) { layer.setStyle({ color: "red", weight: 8, opacity: 1 }); if (layer.bringToFront) layer.bringToFront(); if (layer.getBounds) map.fitBounds(layer.getBounds(), { padding: [50, 50] }); layer.openPopup(); } else { layer.setStyle({ color: "#ff7800", opacity: 0.8, weight: 4 }); } }); }; dropdown.appendChild(item); }); } else { dropdown.style.display = 'none'; } }
